Supabase SQL Setup Script

Copy and run this SQL inside your Supabase SQL Editor to create the inventory table matching all app columns: 

-- 1. Create Inventory Table with UUID Primary Key
create table if not exists public.inventory (
    id uuid default gen_random_uuid() primary key,
    sku text unique not null,
    name text not null,
    category text not null default 'Other',
    quantity integer not null default 0,
    min_threshold integer not null default 5,
    unit_price numeric(10,2) not null default 0.00,
    notes text,
    updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Enable Row Level Security (RLS) & Public Policies
alter table public.inventory enable row level security;

drop policy if exists "Allow public access" on public.inventory;
create policy "Allow public access" 
    on public.inventory 
    for all 
    using (true) 
    with check (true);

-- 3. Enable Supabase Realtime WebSocket Broadcast
alter publication supabase_realtime add table public.inventory;
Run this once in Supabase Dashboard -> SQL Editor
